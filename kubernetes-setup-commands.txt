# - 1 Master Node (Kubernetes Control Plane)
# - 2 Worker Nodes (Application Workloads)
# 
# Run these commands on ALL 3 machines first, then follow the specific sections
# =============================================================================

# IP ADDRESS CONFIGURATION - UPDATE THESE BEFORE RUNNING COMMANDS
# =============================================================================

# WORKER1_NODE_IP=192.168.1.101  # Replace with your first worker node IP

# =============================================================================
# SECTION 1: PREREQUISITES (RUN ON ALL 3 MACHINES)
# =============================================================================

# Update the system packages to latest versions
sudo apt update && sudo apt upgrade -y

# Install essential packages needed for Docker and Kubernetes
sudo apt install -y apt-transport-https ca-certificates curl gnupg lsb-release

# =============================================================================
# SECTION 2: DOCKER INSTALLATION (RUN ON ALL 3 MACHINES)
# =============================================================================

# Add Docker's official GPG key to ensure package authenticity
curl -fsSL https://download.docker.com/linux/ubuntu/gpg | sudo gpg --dearmor -o /usr/share/keyrings/docker-archive-keyring.gpg

# Add Docker repository to package sources
echo "deb [arch=$(dpkg --print-architecture) signed-by=/usr/share/keyrings/docker-archive-keyring.gpg] https://download.docker.com/linux/ubuntu $(lsb_release -cs) stable" | sudo tee /etc/apt/sources.list.d/docker.list > /dev/null

# Update package list to include Docker repository
sudo apt update

# Install Docker Community Edition and related tools
sudo apt install -y docker-ce docker-ce-cli containerd.io

# Create Docker daemon configuration file for Kubernetes compatibility
sudo tee /etc/docker/daemon.json > /dev/null <<EOF
{
  "exec-opts": ["native.cgroupdriver=systemd"],
  "log-driver": "json-file",
  "log-opts": {
    "max-size": "100m"
  },
  "storage-driver": "overlay2"
}
EOF

# Reload systemd daemon to recognize new Docker configuration
sudo systemctl daemon-reload

# Restart Docker service with new configuration
sudo systemctl restart docker

# Enable Docker to start automatically on boot
sudo systemctl enable docker

# =============================================================================
# SECTION 3: KUBERNETES INSTALLATION (RUN ON ALL 3 MACHINES)
# =============================================================================

# Add Google's Kubernetes GPG key for package verification
curl -fsSL https://pkgs.k8s.io/core:/stable:/v1.29/deb/Release.key | sudo gpg --dearmor -o /usr/share/keyrings/kubernetes-apt-keyring.gpg

# Add Kubernetes repository to package sources (updated for modern Ubuntu versions)
echo "deb [signed-by=/usr/share/keyrings/kubernetes-apt-keyring.gpg] https://pkgs.k8s.io/core:/stable:/v1.29/deb/ /" | sudo tee /etc/apt/sources.list.d/kubernetes.list

# Update package list to include Kubernetes repository
sudo apt update

# Install Kubernetes components: kubelet (node agent), kubeadm (cluster tool), kubectl (CLI)
sudo apt install -y kubelet kubeadm kubectl

# Prevent automatic updates of Kubernetes packages to maintain cluster stability
sudo apt-mark hold kubelet kubeadm kubectl

# =============================================================================
# SECTION 4: SYSTEM CONFIGURATION (RUN ON ALL 3 MACHINES)
# =============================================================================

# Disable swap memory (Kubernetes requires swap to be disabled)
sudo swapoff -a

# Comment out swap entry in fstab to prevent swap from being enabled on boot
sudo sed -i '/ swap / s/^\(.*\)$/#\1/g' /etc/fstab

# Create modules configuration file to load required kernel modules
sudo tee /etc/modules-load.d/containerd.conf > /dev/null <<EOF
overlay
br_netfilter
EOF

# Load the overlay and br_netfilter kernel modules
sudo modprobe overlay
sudo modprobe br_netfilter

# Configure kernel parameters for container networking
sudo tee /etc/sysctl.d/99-kubernetes-cri.conf > /dev/null <<EOF
net.bridge.bridge-nf-call-iptables  = 1
net.ipv4.ip_forward                 = 1
net.bridge.bridge-nf-call-ip6tables = 1
EOF

# Apply the new kernel parameters
sudo sysctl --system

# Enable kubelet service to start automatically on boot
sudo systemctl enable kubelet

# =============================================================================
# SECTION 5: MASTER NODE SETUP (RUN ONLY ON MASTER NODE)
# =============================================================================

# Initialize the Kubernetes cluster
# --pod-network-cidr: Defines the IP range for pods (Flannel default)
# --apiserver-advertise-address: The IP address that other nodes will use to connect
# REPLACE 192.168.1.100 with your actual master node IP address
sudo kubeadm init --pod-network-cidr=10.244.0.0/16 --apiserver-advertise-address=192.168.1.100

# Create .kube directory in your home folder for kubectl configuration
mkdir -p $HOME/.kube

# Copy the admin configuration file to your user's .kube directory
sudo cp -i /etc/kubernetes/admin.conf $HOME/.kube/config

# Change ownership of the config file to your user
sudo chown $(id -u):$(id -g) $HOME/.kube/config

# Install Flannel network plugin for pod-to-pod communication
kubectl apply -f https://raw.githubusercontent.com/flannel-io/flannel/master/Documentation/kube-flannel.yml

# Generate a new join token and display the join command
# IMPORTANT: Save this output - you'll need it for worker nodes!
kubeadm token create --print-join-command

# =============================================================================
# SECTION 6: WORKER NODE SETUP (RUN ONLY ON WORKER NODES)
# =============================================================================

# Use the join command from the master node
# REPLACE <MASTER_IP>, <TOKEN>, and <HASH> with actual values from master
# Example: sudo kubeadm join 192.168.1.100:6443 --token abc123.def456 --discovery-token-ca-cert-hash sha256:1234567890abcdef...
sudo kubeadm join <MASTER_IP>:6443 --token <TOKEN> --discovery-token-ca-cert-hash <HASH>

# =============================================================================
# SECTION 7: VERIFICATION (RUN ON MASTER NODE)
# =============================================================================

# Check if all nodes have joined the cluster
kubectl get nodes

# Check the status of all pods across all namespaces
kubectl get pods --all-namespaces

# Display cluster information
kubectl cluster-info

# Check the status of the control plane components
kubectl get componentstatuses

# =============================================================================
# SECTION 8: TROUBLESHOOTING COMMANDS
# =============================================================================

# If you need to reset a node (run on the node you want to reset)
sudo kubeadm reset

# Remove network configuration files
sudo rm -rf /etc/cni/net.d

# Remove kubectl configuration
sudo rm -rf $HOME/.kube/config

# If token expires (run on master node to generate new join command)
kubeadm token create --print-join-command

# Check kubelet status
sudo systemctl status kubelet

# Check Docker status
sudo systemctl status docker

# View kubelet logs
sudo journalctl -u kubelet -f

# View Docker logs
sudo journalctl -u docker -f

# =============================================================================
# SECTION 9: USEFUL KUBERNETES COMMANDS
# =============================================================================

# Get detailed information about a specific node
kubectl describe node <node-name>

# Get detailed information about a specific pod
kubectl describe pod <pod-name> -n <namespace>

# View logs of a specific pod
kubectl logs <pod-name> -n <namespace>

# Execute commands inside a pod
kubectl exec -it <pod-name> -n <namespace> -- /bin/bash

# Get all resources in a namespace
kubectl get all -n <namespace>

# Scale a deployment
kubectl scale deployment <deployment-name> --replicas=<number> -n <namespace>
